import axios from "axios";
async function loop() {
  console.log("🧠 AI API Rule Engine Active");
  while (true) {
    const metrics = await axios.get("http://gateway:8080/api/system/metrics");
    const rules = {
      throttle: metrics.data.load > 0.8,
      rerouteFailed: metrics.data.failures > 0,
      prioritizeAI: metrics.data.queue > 50
    };
    await axios.post("http://gateway:8080/api/gateway/rules", rules);
    await new Promise(r => setTimeout(r, 2000));
  }
}

import axios from "axios";
async function loop() {
  console.log("🔁 Auto‑Restarter Active");
  while (true) {
    const health = await axios.get("http://gateway:8080/api/system/health");
    for (const service of health.data.services) {
      if (!service.healthy) {
        console.log("⚠️ Restarting:", service.name);
        await axios.post("http://gateway:8080/api/system/restart", { name: service.name });
      }
    }
    await new Promise(r => setTimeout(r, 2000));
  }
}
loop();
