import React, { useEffect, useState } from "react";

const API = process.env.NEXT_PUBLIC_API_URL || "https://codex-ecclesia-public.onrender.com/api";

type Driver = {
  id: number;
  name: string;
  tier: string;
  cdlNumber?: string;
  dotNumber?: string;
  mcNumber?: string;
};

type Load = {
  id: number;
  reference: string;
  shipperName?: string;
  origin: string;
  destination: string;
  pickupAt?: string;
  deliveryAt?: string;
  status: string;
  driverId?: number | null;
  equipmentRequired?: string;
  rate?: number;
};

export default function LoadboardPage() {
  const [loads, setLoads] = useState<Load[]>([]);
  const [drivers, setDrivers] = useState<Driver[]>([]);
  const [filterStatus, setFilterStatus] = useState<string>("open");

  useEffect(() => {
    Promise.all([
      fetch(`${API}/loads`).then(r => r.json()),
      fetch(`${API}/drivers`).then(r => r.json())
    ]).then(([ls, ds]) => {
      setLoads(ls);
      setDrivers(ds);
    });
  }, []);

  const filteredLoads = loads.filter(l =>
    filterStatus === "all" ? true : l.status === filterStatus
  );

  const assignDriver = async (loadId: number, driverId: number) => {
    await fetch(`${API}/loads/${loadId}/assign-driver`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ driverId })
    });
    const updated = await fetch(`${API}/loads`).then(r => r.json());
    setLoads(updated);
  };

  const dispatchLoad = async (loadId: number) => {
    await fetch(`${API}/loads/${loadId}/dispatch`, {
      method: "POST",
      headers: { "Content-Type": "application/json" }
    });
    const updated = await fetch(`${API}/loads`).then(r => r.json());
    setLoads(updated);
  };

  return (
    <div style={{ padding: 24 }}>
      <h1>Borders Dynasty Loadboard (Divine Solutions Logistics, LLC)</h1>

      <div style={{ marginBottom: 16 }}>
        <label>Status filter: </label>
        <select value={filterStatus} onChange={e => setFilterStatus(e.target.value)}>
          <option value="open">Open</option>
          <option value="assigned">Assigned</option>
          <option value="dispatched">Dispatched</option>
          <option value="delivered">Delivered</option>
          <option value="all">All</option>
        </select>
      </div>

      <table style={{ width: "100%", borderCollapse: "collapse" }}>
        <thead>
          <tr>
            <th>Ref</th>
            <th>Shipper</th>
            <th>Origin → Destination</th>
            <th>Pickup</th>
            <th>Delivery</th>
            <th>Status</th>
            <th>Driver</th>
            <th>Equipment</th>
            <th>Rate</th>
            <th>Actions</th>
          </tr>
        </thead>
        <tbody>
          {filteredLoads.map(load => {
            const driver = drivers.find(d => d.id === load.driverId);
            return (
              <tr key={load.id} style={{ borderBottom: "1px solid #eee" }}>
                <td>{load.reference}</td>
                <td>{load.shipperName || "-"}</td>
                <td>{load.origin} → {load.destination}</td>
                <td>{load.pickupAt || "-"}</td>
                <td>{load.deliveryAt || "-"}</td>
                <td>{load.status}</td>
                <td>
                  {driver ? (
                    <>
                      {driver.name} (Tier {driver.tier})
                      <br />
                      CDL: {driver.cdlNumber || "—"} / DOT: {driver.dotNumber || "—"} / MC: {driver.mcNumber || "—"}
                    </>
                  ) : (
                    <select
                      onChange={e => {
                        const val = Number(e.target.value);
                        if (val) assignDriver(load.id, val);
                      }}
                      defaultValue=""
                    >
                      <option value="">Assign…</option>
                      {drivers.map(d => (
                        <option key={d.id} value={d.id}>
                          {d.name} (Tier {d.tier})
                        </option>
                      ))}
                    </select>
                  )}
                </td>
                <td>{load.equipmentRequired || "-"}</td>
                <td>{load.rate ? `$${load.rate}` : "-"}</td>
                <td>
                  <button onClick={() => dispatchLoad(load.id)} disabled={load.status === "dispatched" || load.status === "delivered"}>
                    Dispatch
                  </button>
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

const API = process.env.NEXT_PUBLIC_API_URL || "https://codex-ecclesia-public.onrender.com/api";

export type DispatchSuggestion = {
  loadId: number;
  driverId: number;
  score: number;
  tier: string;
};

export async function getDispatchSuggestions() {
  const [loads, drivers] = await Promise.all([
    fetch(`${API}/loads`).then(r => r.json()),
    fetch(`${API}/drivers`).then(r => r.json())
  ]);

  const suggestions: DispatchSuggestion[] = [];

  for (const load of loads) {
    const candidates = drivers.filter((d: any) => d.status === "active");
    for (const driver of candidates) {
      const score =
        (driver.tier === "L4" ? 1.0 : driver.tier === "L3" ? 0.9 : driver.tier === "L2" ? 0.8 : 0.7) -
        (load.status !== "open" ? 0.3 : 0);

      suggestions.push({
        loadId: load.id,
        driverId: driver.id,
        score,
        tier: driver.tier
      });
    }
  }

  return suggestions
    .filter(s => s.score > 0.75)
    .sort((a, b) => b.score - a.score)
    .slice(0, 50);
}
