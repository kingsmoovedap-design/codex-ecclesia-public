contract ReserveClaimToken is

    // Events
    event ReserveMint(address indexed to, uint256 amount);
    event ReserveBurn(address indexed from, uint256 amount);
    event BlacklistUpdated(address indexed account, bool status);

    /// @custom:oz-upgrades-unsafe-allow constructor
    constructor(address _lzEndpoint) OFTCore(_lzEndpoint) {
        _disableInitializers();
    }

    function initialize(address treasury, address governance) public initializer {
        __ERC20_init("Borders Reserve Claim", "BRC");
        __ERC20Pausable_init();
        __AccessControl_init();
        __Ownable_init(treasury);

        _grantRole(DEFAULT_ADMIN_ROLE, treasury);
        _grantRole(TREASURY_ROLE, treasury);
        _grantRole(GOVERNANCE_ROLE, governance);
        _grantRole(COMPLIANCE_ROLE, treasury);

    // Compliance
    // -----------------------------

        external
        onlyRole(COMPLIANCE_ROLE)
    {
        _blacklist[account] = status;
        emit BlacklistUpdated(account, status);
    }

    function isBlacklisted(address account) public view returns (bool) {
        return _blacklist[account];
    }

    // Treasury Mint/Burn
    // -----------------------------

    function mint(address to, uint256 amount)
        external
        onlyRole(TREASURY_ROLE)
    {
        require(!_blacklist[to], "BRC: blacklisted");
        _mint(to, amount);
        emit ReserveMint(to, amount);
    }

        external
        onlyRole(TREASURY_ROLE)

        require(!_blacklist[from], "BRC: blacklisted");
        _burn(from, amount);
        emit ReserveBurn(from, amount);

        address from,
        uint16,
        bytes32,
        uint256 amount

        _burn(from, amount);
        return amount;

        uint16,
        bytes32,
        address to,
        uint256 amount

        require(!_blacklist[to], "BRC: blacklisted");
        _mint(to, amount);

    // Transfer Hook
    // -----------------------------

    function _update(address from, address to, uint256 value)
        internal
        override
    {
        require(!_blacklist[from] && !_blacklist[to], "BRC: blacklisted");
        super._update(from, to, value);
    }

    // -----------------------------
    // Compliance Marker
    // -----------------------------

    function verifyReserveClaim()
        external
        pure
        returns (string memory)
    {
        return "Borders Dynasty Nation — Reserve Claim Token — Omnichain OFT-v2";
    }
}

    Features:
    - ERC20 sweat equity token

    - Vesting schedule (linear or cliff)

    - Upgradeable architecture
*/

import "@openzeppelin/contracts-upgradeable/token/ERC20/extensions/ERC20PausableUpgradeable.sol";
import "@openzeppelin/contracts-upgradeable/access/AccessControlUpgradeable.sol";
import "@openzeppelin/contracts-upgradeable/proxy/utils/Initializable.sol";
import "@openzeppelin/contracts-upgradeable/access/OwnableUpgradeable.sol";

import "@layerzerolabs/oft-evm/contracts/OFTCore.sol";

contract SweatEquityToken is
    Initializable,
    ERC20PausableUpgradeable,
    AccessControlUpgradeable,
    OwnableUpgradeable,
    OFTCore
{
    // Roles
    bytes32 public constant TREASURY_ROLE = keccak256("TREASURY_ROLE");
    bytes32 public constant GOVERNANCE_ROLE = keccak256("GOVERNANCE_ROLE");
    bytes32 public constant COMPLIANCE_ROLE = keccak256("COMPLIANCE_ROLE");

    // Blacklist
    mapping(address => bool) private _blacklist;

    // Vesting
    struct VestingInfo {
        uint256 totalAmount;
        uint256 claimedAmount;
        uint256 startTimestamp;
        uint256 cliffTimestamp;
        uint256 vestingDuration;
    }

    mapping(address => VestingInfo) public vesting;

    // Soulbound mode
    bool public soulbound;

    // Events
    event SweatEquityIssued(address indexed to, uint256 amount);
    event SweatEquityClaimed(address indexed holder, uint256 amount);
    event BlacklistUpdated(address indexed account, bool status);
    event SoulboundModeUpdated(bool enabled);

    /// @custom:oz-upgrades-unsafe-allow constructor
    constructor(address _lzEndpoint) OFTCore(_lzEndpoint) {
        _disableInitializers();
    }

    function initialize(
        address treasury,
        address governance,
        bool _soulbound
    ) public initializer {
        __ERC20_init("Borders Sweat Equity", "BSE");
        __ERC20Pausable_init();
        __AccessControl_init();
        __Ownable_init(treasury);

        _grantRole(DEFAULT_ADMIN_ROLE, treasury);
        _grantRole(TREASURY_ROLE, treasury);
        _grantRole(GOVERNANCE_ROLE, governance);
        _grantRole(COMPLIANCE_ROLE, treasury);

        soulbound = _soulbound;
    }

    // -----------------------------
    // Governance Controls
    // -----------------------------

    function setSoulbound(bool enabled)
        external
        onlyRole(GOVERNANCE_ROLE)
    {
        soulbound = enabled;
        emit SoulboundModeUpdated(enabled);
    }

    // -----------------------------
    // Compliance
    // -----------------------------

    function setBlacklist(address account, bool status)
        external
        onlyRole(COMPLIANCE_ROLE)
    {
        _blacklist[account] = status;
        emit BlacklistUpdated(account, status);
    }

    function isBlacklisted(address account) public view returns (bool) {
        return _blacklist[account];
    }

    // -----------------------------
    // Vesting Logic
    // -----------------------------

    function createVesting(
        address beneficiary,
        uint256 amount,
        uint256 start,
        uint256 cliff,
        uint256 duration
    ) external onlyRole(TREASURY_ROLE) {
        require(vesting[beneficiary].totalAmount == 0, "BSE: vesting exists");

        vesting[beneficiary] = VestingInfo({
            totalAmount: amount,
            claimedAmount: 0,
            startTimestamp: start,
            cliffTimestamp: cliff,
            vestingDuration: duration
        });

        _mint(address(this), amount);
        emit SweatEquityIssued(beneficiary, amount);
    }

    function claimVested() external {
        VestingInfo storage v = vesting[msg.sender];
        require(block.timestamp >= v.cliffTimestamp, "BSE: cliff not reached");

        uint256 vested = _vestedAmount(msg.sender);
        uint256 claimable = vested - v.claimedAmount;

        require(claimable > 0, "BSE: nothing to claim");

        v.claimedAmount += claimable;
        _transfer(address(this), msg.sender, claimable);

        emit SweatEquityClaimed(msg.sender, claimable);
    }

    function _vestedAmount(address account) internal view returns (uint256) {
        VestingInfo memory v = vesting[account];

        if (block.timestamp < v.cliffTimestamp) return 0;
        if (block.timestamp >= v.startTimestamp + v.vestingDuration)
            return v.totalAmount;

        uint256 elapsed = block.timestamp - v.startTimestamp;
