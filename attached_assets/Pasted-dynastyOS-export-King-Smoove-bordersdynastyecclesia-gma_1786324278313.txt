8. SovereignRouter.ts
Unified call surface for ALL sovereign modules.
ts
// spine/SovereignRouter.ts

import { SovereignTelemetryBus } from './SovereignTelemetryBus';

export class SovereignRouter {
  constructor(private modules: Record<string, any>) {}

  async call(module: string, action: string, payload: any) {
    const mod = this.modules[module];
    if (!mod) return { ok: false, error: 'Unknown sovereign module' };

    const result = await mod.handle(action, payload);

    SovereignTelemetryBus.publish({
      id: `sov-evt-${Date.now()}`,
      domain: module,
      type: `${module}.${action}`,
      payload: result,
      timestamp: new Date().toISOString()
    });

    return result;
  }
}

// spine/SovereignEconomyEngine.ts

import { SovereignCreditEngine } from './SovereignCreditEngine';

export class SovereignEconomyEngine {
  static snapshot() {
    const all = SovereignCreditEngine.list();
    const byDomain: Record<string, number> = {};
    const byType: Record<string, number> = {};
    let total = 0;

    for (const c of all) {
      total += c.amount;
      byDomain[c.domain] = (byDomain[c.domain] || 0) + c.amount;
      byType[c.type] = (byType[c.type] || 0) + c.amount;
    }

    return {
      totalCredits: total,
      byDomain,
      byType,
      timestamp: new Date().toISOString()
    };
  }
}
