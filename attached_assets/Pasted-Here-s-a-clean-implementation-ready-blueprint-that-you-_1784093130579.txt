// src/services/dispatchAI.ts
import { getOpenLoads } from './marketplaceService';
import { getAvailableDrivers } from './carrierService';

export async function getRecommendedLoads() {
  const loads = await getOpenLoads();
  const drivers = await getAvailableDrivers();

  // simple scoring placeholder — replace with real AI later
  const scored = loads.map(load => {
    const bestDriver = drivers[0]; // pick by equipment, region, etc.
    return {
      load,
      driver: bestDriver,
      score: Math.random(), // risk/fit score
    };
  });

  return scored.sort((a, b) => b.score - a.score);
}

### 5. Full Stripe subscription dashboard

```ts
// src/api/billing/portal.ts
import Stripe from 'stripe';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY);

export default async function handler(req, res) {
  const { customerId } = req.body;

  const session = await stripe.billingPortal.sessions.create({
    customer: customerId,
    return_url: `${process.env.APP_URL}/dashboard`,
  });

  res.json({ url: session.url });
}
```

On the frontend, link a “Manage Subscription” button to this endpoint and redirect to `session.url`.
