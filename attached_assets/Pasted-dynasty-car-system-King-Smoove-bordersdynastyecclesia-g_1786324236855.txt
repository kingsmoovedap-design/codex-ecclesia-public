3. LuxuryOS: Car pricing engine (SCX‑integrated)
ts
// luxury/car/LuxuryCarPricingEngine.ts
import { scxPricingTable } from "../financial/scx-pricing-table.json";

export interface CarBuildConfig {
  baseModelCode: string;
  options: { code: string; priceUSD: number }[];
  rarityScore: number;      // 0–1
  craftsmanshipScore: number; // 0–1
  riskScore: number;        // 0–1
}

export interface CarPricingResult {
  assetCode: string;
  notionalUSD: number;
  finalYieldPercent: number;
  indicativePriceUSD: number;
}

export class LuxuryCarPricingEngine {
  static price(build: CarBuildConfig): CarPricingResult {
    const baseYield = scxPricingTable.inputs.baseYieldPercent;
    const liquidityFactor = scxPricingTable.inputs.liquidityFactors.medium;

    const optionsTotal = build.options.reduce(
      (sum, o) => sum + o.priceUSD,
      0
    );

    const notionalUSD = optionsTotal * (1 + build.craftsmanshipScore * 0.3);

    const riskPremiumPercent = build.riskScore * 2.0;
    const modernizationPremiumPercent = build.rarityScore * 1.5;

    const finalYieldPercent =
      baseYield * liquidityFactor +
      riskPremiumPercent -
      modernizationPremiumPercent;

    const indicativePriceUSD =
      notionalUSD *
      (1 - (finalYieldPercent - baseYield) / 100);

    return {
      assetCode: `LUX-CAR-${build.baseModelCode}`,
      notionalUSD,
      finalYieldPercent,
      indicativePriceUSD
    };
  }
}

// luxury/car/LuxuryCarProvenanceLedger.ts
export interface CarProvenanceEntry {
  id: string;
  buildId: string;
  event:
    | "BUILD_CREATED"
    | "CONFIGURED"
    | "PRICED"
    | "DEPOSIT_CONFIRMED"
    | "IN_PRODUCTION"
    | "COMPLETED"
    | "DELIVERED"
    | "TRANSFERRED";
  actorId: string;
  timestamp: string;
  notes?: string;
}

const ledger: CarProvenanceEntry[] = [];

export class LuxuryCarProvenanceLedger {
  static record(entry: Omit<CarProvenanceEntry, "id" | "timestamp">) {
    const full: CarProvenanceEntry = {
      id: `CAR-PROV-${Date.now()}`,
      timestamp: new Date().toISOString(),
      ...entry
    };
    ledger.push(full);
    return full;
  }

  static listByBuild(buildId: string): CarProvenanceEntry[] {
    return ledger.filter(e => e.buildId === buildId);
  }
}
