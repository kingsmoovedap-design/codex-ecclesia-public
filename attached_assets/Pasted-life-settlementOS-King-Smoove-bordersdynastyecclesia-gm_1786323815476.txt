import { DYNASTY_POLICIES, DynastyPolicy } from "./wealthPolicyModel";

export interface DynastyPortfolioSnapshot {
  totalFaceAmount: number;
  totalAcquisitionCost: number;
  totalPremiumAnnual: number;
  activeCount: number;
  maturedCount: number;
}

export function getDynastyPortfolioSnapshot(): DynastyPortfolioSnapshot {
  const active = DYNASTY_POLICIES.filter(p => p.status === "active");
  const matured = DYNASTY_POLICIES.filter(p => p.status === "matured");

  return {
    totalFaceAmount: active.reduce((s, p) => s + p.faceAmount, 0),
    totalAcquisitionCost: active.reduce((s, p) => s + p.acquisitionCost, 0),
    totalPremiumAnnual: active.reduce((s, p) => s + p.premiumSchedule.amount *
      (p.premiumSchedule.frequency === "monthly" ? 12 :
       p.premiumSchedule.frequency === "quarterly" ? 4 : 1), 0),
    activeCount: active.length,
    maturedCount: matured.length,
  };
}

import { DynastyPolicy } from "./wealthPolicyModel";

export interface ActuarialProjection {
  policyId: string;
  expectedMaturityYear: number;
  expectedReturnMultiple: number;
  impliedIRR: number;
}

export function projectPolicyActuarial(policy: DynastyPolicy): ActuarialProjection {
  const years = policy.expectedMaturityYear - new Date().getFullYear();
  const totalPremium = years *
    policy.premiumSchedule.amount *
    (policy.premiumSchedule.frequency === "monthly" ? 12 :
     policy.premiumSchedule.frequency === "quarterly" ? 4 : 1);

  const totalCost = policy.acquisitionCost + totalPremium;
  const multiple = policy.faceAmount / totalCost;
  const impliedIRR = Math.pow(multiple, 1 / Math.max(years, 1)) - 1;

  return {
    policyId: policy.id,
    expectedMaturityYear: policy.expectedMaturityYear,
    expectedReturnMultiple: multiple,
    impliedIRR,
  };
}

import { DynastyPolicy } from "./wealthPolicyModel";

export interface PolicyRiskScore {
  policyId: string;
  premiumRisk: number;
  longevityRisk: number;
  concentrationRisk: number;
  compositeRisk: number;
}

export function evaluatePolicyRisk(policy: DynastyPolicy, portfolioSize: number): PolicyRiskScore {
  const premiumRisk = policy.premiumSchedule.amount > 5000 ? 0.7 : 0.3;
  const longevityRisk = policy.expectedMaturityYear - new Date().getFullYear() > 15 ? 0.6 : 0.3;
  const concentrationRisk = policy.faceAmount / Math.max(portfolioSize, 1) > 0.2 ? 0.8 : 0.3;

  const composite = (premiumRisk + longevityRisk + concentrationRisk) / 3;

  return {
    policyId: policy.id,
    premiumRisk,
    longevityRisk,
    concentrationRisk,
    compositeRisk: composite,
  };
}

import { MarketPolicy } from "./marketPolicyModel";

export interface UnderwritingResult {
  policyId: string;
  approved: boolean;
  riskClass: "low" | "medium" | "high";
  notes?: string;
}

export function underwritePolicy(policy: MarketPolicy): UnderwritingResult {
  const score = policy.healthScore;
  const age = policy.insuredAge;

  if (score > 80 && age > 75) {
    return { policyId: policy.id, approved: true, riskClass: "low" };
  }
  if (score > 60 && age > 70) {
    return { policyId: policy.id, approved: true, riskClass: "medium" };
  }
  return { policyId: policy.id, approved: false, riskClass: "high", notes: "Does not meet underwriting criteria" };
}

    evaluatePolicyRisk(p, DYNASTY_POLICIES.length)
  );

  const commercialRisk = MARKET_POLICIES.map(p =>
    underwritePolicy(p)
  );

  const composite = {
    dynastyCompositeRisk:
      dynastyRisk.reduce((s, r) => s + r.compositeRisk, 0) /
      Math.max(dynastyRisk.length, 1),

    commercialApprovalRate:
      commercialRisk.filter(r => r.approved).length /
      Math.max(commercialRisk.length, 1),

    combinedSignal:
      dynastyRisk.length > 0 && commercialRisk.length > 0
        ? "DUAL-TRACK ACTIVE"
        : "PARTIAL"
  };

  res.json({ dynastyRisk, commercialRisk, composite });
});

// lifeSettlementMarket/executionEngine.ts
import { MarketPolicy, MARKET_POLICIES } from "./marketPolicyModel";
import { generatePriceQuote } from "./pricingEngine";
import { recordLedgerEntry } from "../ledger/ledger";

export interface ExecutionOrder {
  policyId: string;
  buyerId: string;
  type: "BUY";
}

export interface ExecutionResult {
  policyId: string;
  buyerId: string;
  executedPrice: number;
  status: "EXECUTED" | "FAILED";
  reason?: string;
}

export function executeOrder(order: ExecutionOrder): ExecutionResult {
  const policy = MARKET_POLICIES.find(p => p.id === order.policyId);
  if (!policy || policy.status !== "listed") {
    return { policyId: order.policyId, buyerId: order.buyerId, executedPrice: 0, status: "FAILED", reason: "Not listed" };
  }

  const quote = generatePriceQuote(policy);
  policy.status = "sold";

  recordLedgerEntry({
    id: `ls-mkt-led-${Date.now()}`,
    sovereignId: "DELAWARE-LLC",
    cycleId: `ls-mkt-cycle-${Date.now()}`,
    nodeId: policy.id,
    action: "LIFE_SETTLEMENT_SALE",
    cost: -quote.bidPrice,
    impact: policy.faceAmount - quote.bidPrice,
    timestamp: new Date().toISOString(),
  });

  return {
    policyId: policy.id,
    buyerId: order.buyerId,
    executedPrice: quote.bidPrice,
    status: "EXECUTED",
  };
}

3. Premium Servicing Engine
Purpose: Track and advance premium payments for both dynasty and commercial rails.

// lifeSettlement/premiumServicingEngine.ts
import { DYNASTY_POLICIES } from "./wealthPolicyModel";
import { MARKET_POLICIES } from "../lifeSettlementMarket/marketPolicyModel";
import { recordLedgerEntry } from "../ledger/ledger";

export interface PremiumServicingResult {
  dynastyPremiumTotal: number;
  commercialPremiumTotal: number;
  timestamp: string;
}

export function runPremiumServicingCycle(): PremiumServicingResult {
  const dynastyPremiumTotal = DYNASTY_POLICIES.reduce((s, p) => {
    return s + p.premiumSchedule.amount;
  }, 0);

  const commercialPremiumTotal = MARKET_POLICIES.reduce((s, p) => {
    return s + p.premiumAnnual / 12; // monthly servicing
  }, 0);

  recordLedgerEntry({
    id: `ls-prem-led-${Date.now()}`,
    sovereignId: "NEVADA-DYNASTY",
    cycleId: `ls-prem-cycle-${Date.now()}`,
    nodeId: "DYNASTY-PREMIUM",
    action: "DYNASTY_PREMIUM_SERVICING",
    cost: dynastyPremiumTotal,
    impact: -dynastyPremiumTotal,
    timestamp: new Date().toISOString(),
  });

  recordLedgerEntry({
    id: `ls-prem-led-${Date.now() + 1}`,
    sovereignId: "DELAWARE-LLC",
    cycleId: `ls-prem-cycle-${Date.now()}`,
    nodeId: "COMMERCIAL-PREMIUM",
    action: "COMMERCIAL_PREMIUM_SERVICING",
    cost: commercialPremiumTotal,
    impact: -commercialPremiumTotal,
    timestamp: new Date().toISOString(),
  });

  return {
    dynastyPremiumTotal,
    commercialPremiumTotal,
    timestamp: new Date().toISOString(),

// treasury/liquidityEngine.ts

export interface LiquidityEvent {
  id: string;
  source: "DYNASTY" | "COMMERCE" | "TREASURY";
  type: "INFLOW" | "OUTFLOW";
  amount: number;
  timestamp: string;
  note?: string;
}

export interface LiquiditySnapshot {
  snapshotId: string;
  totalLiquidity: number;
  dynastyLiquidity: number;
  commerceLiquidity: number;
  treasuryLiquidity: number;
  events: LiquidityEvent[];
  timestamp: string;
}

const LIQUIDITY_EVENTS: LiquidityEvent[] = [];

let dynastyLiquidity = 0;
let commerceLiquidity = 0;
let treasuryLiquidity = 0;

export function recordLiquidityEvent(event: LiquidityEvent): void {
  LIQUIDITY_EVENTS.push(event);

  const signedAmount = event.type === "INFLOW" ? event.amount : -event.amount;

  if (event.source === "DYNASTY") dynastyLiquidity += signedAmount;
  if (event.source === "COMMERCE") commerceLiquidity += signedAmount;
  if (event.source === "TREASURY") treasuryLiquidity += signedAmount;
}

export function getLiquiditySnapshot(): LiquiditySnapshot {
  const totalLiquidity = dynastyLiquidity + commerceLiquidity + treasuryLiquidity;

  return {
    snapshotId: `liq-snap-${Date.now()}`,
    totalLiquidity,
    dynastyLiquidity,
    commerceLiquidity,
    treasuryLiquidity,
    events: LIQUIDITY_EVENTS.slice(-100), // last 100 events
    timestamp: new Date().toISOString(),
  };
}

Simple scaling function
ts
// models/commerceScalingModel.ts

export interface CommerceScalingInput {
  policiesPerYear: number;
  avgTransactionValue: number;      // purchase price per policy
  marketplaceFeeRate: number;       // e.g. 0.015 = 1.5%
  underwritingFeePerPolicy: number; // e.g. 500
  servicingFeePerPolicyPerMonth: number; // e.g. 35
  conversionRate: number;           // e.g. 0.7 = 70% of listed policies close
  operatingCostPerYear: number;     // fixed + variable
}

export interface CommerceScalingOutput {
  grossRevenue: number;
  netRevenue: number;
  breakdown: {
    marketplaceFees: number;
    underwritingFees: number;
    servicingFees: number;
  };
}

export function projectCommerceScaling(input: CommerceScalingInput): CommerceScalingOutput {
  const closedPolicies = input.policiesPerYear * input.conversionRate;

  const marketplaceFees =
    closedPolicies * input.avgTransactionValue * input.marketplaceFeeRate;

  const underwritingFees =
    input.policiesPerYear * input.underwritingFeePerPolicy;

  const servicingFees =
    closedPolicies * input.servicingFeePerPolicyPerMonth * 12;

  const grossRevenue = marketplaceFees + underwritingFees + servicingFees;
  const netRevenue = grossRevenue - input.operatingCostPerYear;

  return {
    grossRevenue,
    netRevenue,
    breakdown: {
      marketplaceFees,
      underwritingFees,
      servicingFees,
    },
  };
}
