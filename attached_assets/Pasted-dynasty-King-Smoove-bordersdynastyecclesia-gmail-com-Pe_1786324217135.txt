export interface YieldCurve {
  id: string;
  sovereignId: string;

  price(inst: IMXInstrument): number;

export class YieldCurveEngine {
  private curves = new Map<string, YieldCurve>();

  registerCurve(curve: YieldCurve) {
    this.curves.set(curve.id, curve);
  }

  async getCurve(id: string): Promise<YieldCurve> {
    const curve = this.curves.get(id);
    if (!curve) throw new Error("curve not found");
    return curve;
  }

  buildFromTelemetry(telemetry: any): YieldCurve {
    // derive rates from stability, efficiency, risk
    return {
      id: `yc-${Date.now()}`,
      sovereignId: telemetry.sovereignId,
      points: [{ tenorDays: 365, rate: 0.05 }],
      price(inst) {
        const base = inst.notional;
        const r = this.points[0].rate;
        return base / (1 + r);
      }
    };
  }
}
Infrastructure Liquidity Pool engine

// imx/liquidity/LiquidityPoolEngine.ts
export interface LiquidityPool {
  id: string;
  sovereignId: string;
  instrumentType: IMXInstrumentType;
  totalLiquidity: number;
  utilization: number;
}

export class LiquidityPoolEngine {
  private pools = new Map<string, LiquidityPool>();

  registerPool(pool: LiquidityPool) {
    this.pools.set(pool.id, pool);
  }

  async route(inst: IMXInstrument) {
    const pool = [...this.pools.values()].find(
      p => p.sovereignId === inst.sovereignId && p.instrumentType === inst.type
    );
    if (!pool) throw new Error("no pool");
    const available = pool.totalLiquidity * (1 - pool.utilization);
    const filled = Math.min(available, inst.notional);
    pool.utilization = (pool.utilization * pool.totalLiquidity + filled) / pool.totalLiquidity;
    return { poolId: pool.id, filled };
  }
}

// infra/risk/InfrastructureRiskEngine.ts
export interface RiskReport {
  nodeId: string;
  sovereignId: string;
  stability: number;
  congestion: number;
  liquidityRisk: number;
  modernizationRisk: number;
  composite: number;
}

export class InfrastructureRiskEngine {
  compute(graph: UIGraph): RiskReport[] {
    return graph.nodes.map(n => {
      const stability = n.metrics.stability ?? 0.9;
      const congestion = n.metrics.congestion ?? 0.2;
      const liquidityRisk = n.metrics.liquidityRisk ?? 0.1;
      const modernizationRisk = n.metrics.modRisk ?? 0.15;

      const composite =
        0.4 * (1 - stability) +
        0.3 * congestion +
        0.2 * liquidityRisk +
        0.1 * modernizationRisk;

      return {
        nodeId: n.id,
        sovereignId: n.sovereignId,
        stability,
        congestion,
        liquidityRisk,
        modernizationRisk,
        composite
      };
    });
  }
}

export class ExecutionEngine {
  execute(graph: UIGraph, instructions: UpgradeInstruction[]) {
    for (const inst of instructions) {
      const node = graph.nodes.find(n => n.id === inst.nodeId);
      if (!node) continue;

      // apply upgrade parameters
      for (const [k, v] of Object.entries(inst.params)) {
        node.metrics[k] = v;
      }
    }

    return graph;
  }

export interface SovereignMetricSnapshot {
  sovereignId: string;
  stabilityIndex: number;
  congestionIndex: number;
  liquidityIndex: number;
}

export interface CoordinationPlan {
  id: string;
  fromSovereignId: string;
  toSovereignId: string;
  action: "REDISTRIBUTE_LOAD" | "REDISTRIBUTE_LIQUIDITY";
  amount: number;
  reason: string;
}

export class SovereignCoordinationProtocol {
  buildPlan(snapshots: SovereignMetricSnapshot[]): CoordinationPlan[] {
    const plans: CoordinationPlan[] = [];
    const sorted = [...snapshots].sort((a, b) => a.stabilityIndex - b.stabilityIndex);

    const low = sorted[0];
    const high = sorted[sorted.length - 1];

    if (low.stabilityIndex < 0.8 && high.stabilityIndex > 0.95) {
      plans.push({
        id: `coord-${Date.now()}`,
        fromSovereignId: high.sovereignId,
        toSovereignId: low.sovereignId,
        action: "REDISTRIBUTE_LIQUIDITY",
        amount: 0.05,
        reason: "Balance stability across sovereigns"
      });
    }

    return plans;
  }
}

        yieldCurveId: yc.id,
        liquidityPoolId: `pool-${sovereignId}`
      });
    }

    // 9. Safety envelope (risk + transparency + oversight)
    const safetyResult = await this.safety.protect(
      { sovereignId, stabilityIndex: stability },
      { decisions, instructions, federation }
    );

    // 10. Coordination across sovereigns (placeholder)
    const snapshots = [
      { sovereignId, stabilityIndex: stability, congestionIndex: 0.3, liquidityIndex: 0.7 }
      // add other sovereigns here
    ];
    const coordPlans = this.coordProto.buildPlan(snapshots);

    return {
      updatedGraph,
      riskReports,
      stability,
      decisions,
      federation,
      instructions,
      safetyResult,
      coordPlans
    };
  }
}

}
You can gate the runtime API:
ts
// api/middleware/infraAuth.ts
export function requireInfraPermission(action: InfraPermission["action"]) {
  return (req, res, next) => {
    const role: InfraRole = req.user.infraRole;
    if (!can(role, action)) return res.status(403).json({ error: "forbidden" });
    next();
  };
}

# DynastyOS Infrastructure Runtime (DIR)
Technical Handbook (Non‑Political)

The DynastyOS Infrastructure Runtime orchestrates sovereign‑scale modernization
cycles across corridors, logistics, fuel, treasury, and SCX infrastructure.

It wires together:

- Unified Infrastructure Graph (UIG)
- Infrastructure Risk & Stability
- Infrastructure AI Agents
- IMX Marketplace, Yield Curves, Liquidity Pools
- Digital Twin Federation
- Upgrade Compiler & Execution Engine
- Safety Envelope & Sovereign Coordination

---

## 1. Sovereign Configuration

Each sovereign has a config file:

- `config/sovereigns/<sovereignId>.yaml`

Key fields:

- `infrastructure.*` — feature flags (telemetry, twins, IMX, safety, coordination)
- `limits.*` — liquidity, budget, actions per cycle
- `safety.*` — risk thresholds, stability targets, oversight triggers

Infra admins manage these configs and must keep them under version control.

---

## 2. Roles and RBAC

Infra runtime uses four roles:

- **INFRA_ADMIN** — full configuration + execution
- **INFRA_OPERATOR** — run cycles, execute upgrades, view risk/ledger
- **INFRA_AUDITOR** — read‑only risk, ledger, IMX, twins
- **INFRA_VIEWER** — high‑level visibility only

APIs are gated by permissions:

- `RUN_INFRA_CYCLE`
- `VIEW_INFRA_RISK`
- `VIEW_INFRA_LEDGER`
- `CONFIGURE_SOVEREIGN_INFRA`
- `VIEW_TWIN_FEDERATION`
- `VIEW_IMX`
- `EXECUTE_UPGRADES`

---

## 3. Runtime Cycle Overview

For each sovereign:

1. **Load UIG**  
2. **Compute risk + stability**  
3. **AI agents propose modernization actions**  
4. **Build yield curves + register liquidity pools**  
5. **Build digital twin federation**  
6. **Compile decisions into upgrade instructions**  
7. **Execute upgrades and update UIG**  
8. **Record entries in Infrastructure Modernization Ledger (IML)**  
9. **Apply safety envelope (risk, transparency, oversight)**  
10. **Generate sovereign coordination plans**

---

## 4. Operator Responsibilities

### INFRA_ADMIN

- Maintain sovereign configs
- Set risk thresholds and stability targets
- Approve major modernization cycles
- Review coordination plans across sovereigns

### INFRA_OPERATOR

- Run infrastructure cycles for assigned sovereigns
- Monitor risk and stability dashboards
- Execute approved upgrade instructions
- Verify ledger entries and explanations

### INFRA_AUDITOR

- Review risk reports and stability indices
- Inspect ledger for completeness and consistency
- Confirm safety envelope is invoked for high‑risk actions

### INFRA_VIEWER

- Use dashboards for situational awareness
- Provide feedback on clarity and usability

---

## 5. Safety Envelope

All runtime actions are wrapped by the Infrastructure Safety Envelope (ISE):

- **Resilience:** detect telemetry poisoning and anomalies
- **Privacy:** ensure telemetry is anonymized before analysis
- **Transparency:** generate human‑readable explanations
- **Oversight:** enqueue high‑risk decisions for human review

---

## 6. Coordination Protocol

The Sovereign Coordination Protocol:

- Reads stability and congestion metrics per sovereign
- Proposes liquidity or load redistribution plans
- Logs coordination plans for review and execution

Coordination is technical only—no political content is defined here.
