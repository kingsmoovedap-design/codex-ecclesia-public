1. Sovereign Credit Registry (spec + code)
Spec
Purpose: Define, mint, and manage sovereign credit programs and supplies.
Core concepts: program, totalSupply, availableSupply, impactTemplate.
Code
ts
// src/sovereign/SovereignCreditRegistry.ts
import { pg } from "../db/postgres";

export interface SovereignCreditProgram {
  id: string;
  code: string; // DOT_MODERNIZATION, PORT_DIGITALIZATION, etc.
  name: string;
  totalSupply: number;
  availableSupply: number;
  impactTemplate: any;
}

export class SovereignCreditRegistry {
  async listPrograms(): Promise<SovereignCreditProgram[]> {
    const res = await pg.query("SELECT * FROM sovereign_credit_programs");
    return res.rows;
  }

  async createProgram(input: {
    code: string;
    name: string;
    totalSupply: number;
    impactTemplate: any;
  }): Promise<SovereignCreditProgram> {
    const id = `SCP-${Date.now()}`;
    await pg.query(
      `INSERT INTO sovereign_credit_programs
       (id, code, name, total_supply, available_supply, impact_template)
       VALUES ($1,$2,$3,$4,$5,$6)`,
      [id, input.code, input.name, input.totalSupply, input.totalSupply, input.impactTemplate]
    );
    return {
      id,
      code: input.code,
      name: input.name,
      totalSupply: input.totalSupply,
      availableSupply: input.totalSupply,
      impactTemplate: input.impactTemplate
    };
  }

  async allocate(programCode: string, amount: number): Promise<void> {
    await pg.query(
      `UPDATE sovereign_credit_programs
       SET available_supply = available_supply - $1
       WHERE code = $2 AND available_supply >= $1`,
      [amount, programCode]
    );
  }
}

// src/treasury/TreasuryVault.ts
import { pg } from "../db/postgres";

export interface TreasuryVaultState {
  programCode: string;
  creditBalance: number;
  cashBalance: number;
}

export class TreasuryVault {
  async getState(): Promise<TreasuryVaultState[]> {
    const res = await pg.query("SELECT * FROM treasury_vault");
    return res.rows;
  }

  async creditDeposit(programCode: string, amount: number) {
    await pg.query(
      `INSERT INTO treasury_vault (program_code, credit_balance, cash_balance)
       VALUES ($1,$2,0)
       ON CONFLICT (program_code)
       DO UPDATE SET credit_balance = treasury_vault.credit_balance + EXCLUDED.credit_balance`,
      [programCode, amount]
    );
  }

  async cashDeposit(programCode: string, amount: number) {
    await pg.query(
      `INSERT INTO treasury_vault (program_code, credit_balance, cash_balance)
       VALUES ($1,0,$2)
       ON CONFLICT (program_code)
       DO UPDATE SET cash_balance = treasury_vault.cash_balance + EXCLUDED.cash_balance`,
      [programCode, amount]
    );
  }

  async consumeCredits(programCode: string, amount: number) {
    await pg.query(
      `UPDATE treasury_vault
       SET credit_balance = credit_balance - $1
       WHERE program_code = $2 AND credit_balance >= $1`,
      [amount, programCode]
    );
  }
}

3. MarketplaceOS Sovereign Credit Exchange (spec + code)
Spec
Purpose: List credits for sale, handle buy orders, connect to TreasuryVault.
Core concepts: listing, pricePerCredit, programCode, availableAmount.
Code
ts
// src/marketplace/SovereignCreditExchange.ts
import { pg } from "../db/postgres";
import { TreasuryVault } from "../treasury/TreasuryVault";
import { SovereignCreditRegistry } from "../sovereign/SovereignCreditRegistry";

export interface CreditListing {
  id: string;
  programCode: string;
  pricePerCredit: number;
  availableAmount: number;
}

export class SovereignCreditExchange {
  constructor(
    private vault = new TreasuryVault(),
    private registry = new SovereignCreditRegistry()
  ) {}

  async listListings(): Promise<CreditListing[]> {
    const res = await pg.query("SELECT * FROM sovereign_credit_listings");
    return res.rows;
  }

  async createListing(programCode: string, pricePerCredit: number, amount: number) {
    const id = `SCL-${Date.now()}`;
    await this.registry.allocate(programCode, amount);
    await this.vault.creditDeposit(programCode, amount);

    await pg.query(
      `INSERT INTO sovereign_credit_listings
       (id, program_code, price_per_credit, available_amount)
       VALUES ($1,$2,$3,$4)`,
      [id, programCode, pricePerCredit, amount]
    );

    return { id, programCode, pricePerCredit, availableAmount: amount };
  }

  async buyCredits(listingId: string, buyerId: string, amount: number) {
    const listingRes = await pg.query(
      "SELECT * FROM sovereign_credit_listings WHERE id = $1",
      [listingId]
    );
    if (!listingRes.rowCount) throw new Error("listing_not_found");
    const listing = listingRes.rows[0];

    if (listing.available_amount < amount) throw new Error("insufficient_listing_amount");

    const totalPrice = listing.price_per_credit * amount;

    await pg.query(
      `UPDATE sovereign_credit_listings
       SET available_amount = available_amount - $1
       WHERE id = $2`,
      [amount, listingId]
    );

    await this.vault.cashDeposit(listing.program_code, totalPrice);
    await this.vault.consumeCredits(listing.program_code, amount);

    await pg.query(
      `INSERT INTO sovereign_credit_purchases
       (id, buyer_id, program_code, amount, total_price)
       VALUES ($1,$2,$3,$4,$5)`,
      [`SCPUR-${Date.now()}`, buyerId, listing.program_code, amount, totalPrice]
    );
  }
}

4. Liquidity Engine pricing curves for credits
Spec
Purpose: Dynamic pricing based on demand, supply, and impact.
Inputs: basePrice, demandIndex, supplyRatio, impactScore.
Code
ts
// src/marketplace/CreditLiquidityEngine.ts
export interface PricingContext {
  basePrice: number;
  demandIndex: number; // 0–1
  supplyRatio: number; // availableSupply / totalSupply
  impactScore: number; // 0–1
}

export class CreditLiquidityEngine {
  computePrice(ctx: PricingContext): number {
    const demandFactor = 1 + ctx.demandIndex * 0.5; // up to +50%
    const scarcityFactor = ctx.supplyRatio < 0.3 ? 1.3 : 1.0; // scarce → +30%
    const impactFactor = 1 + ctx.impactScore * 0.4; // high impact → +40%

    return ctx.basePrice * demandFactor * scarcityFactor * impactFactor;
  }
}
You can plug this into SovereignCreditExchange.createListing() to set pricePerCredit.

              <div className="flex gap-2 items-center">
                <span className={`w-3 h-3 rounded-full ${congestionColor}`} />
                <span className="text-xs text-slate-300">
                  Congestion {(r.congestionIndex * 100).toFixed(0)}%
                </span>
              </div>
              <div className="flex gap-2 items-center mt-1">
                <span className={`w-3 h-3 rounded-full ${idleColor}`} />
                <span className="text-xs text-slate-300">
                  Idle {(r.idleIndex * 100).toFixed(0)}%
                </span>
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
};
