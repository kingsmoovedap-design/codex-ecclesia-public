// core/global/infrastructure/EventBus.ts
export interface GlobalEventEnvelope<T> {
  id: string;
  type: string;
  source: string;
  version: string;
  timestamp: string;
  level: "LOCAL" | "STATE" | "NATIONAL" | "REGIONAL" | "GLOBAL";
  payload: T;
  meta?: Record<string, any>;
}

type Handler = (event: GlobalEventEnvelope<any>) => Promise<void>;

export class EventBus {
  private static handlers: Record<string, Handler[]> = {};

  static subscribe(eventType: string, handler: Handler) {
    if (!this.handlers[eventType]) this.handlers[eventType] = [];
    this.handlers[eventType].push(handler);
  }

  static async publish<T>(event: GlobalEventEnvelope<T>) {
    const handlers = this.handlers[event.type] || [];
    for (const h of handlers) await h(event);
  }
}

        version: "1.0",
        timestamp: new Date().toISOString(),
        level: event.level,
        payload: { corridorId: m.corridorId, anomaly }
      });
    }
  }
}

import {
  SovereignModernizationLifecycleEngine,
  LifecycleStepId
} from "../lifecycle/SovereignModernizationLifecycleEngine";
import { ModernizationPlanRepository } from "./ModernizationPlanRepository";

export interface ModernizationPlan {
  id: string;
  corridorId: string;
  budgetUSD: number;
  expectedROI: number;
  steps: LifecycleStepId[];

export class AutonomousModernizationPlanner {
  constructor(
    private gmig: GMIG,
    private lifecycleEngine: SovereignModernizationLifecycleEngine,
    private repo: ModernizationPlanRepository
  ) {}

  async generateQuarterPlan(): Promise<ModernizationPlan | null> {
    const bestCorridor = await this.gmig.queryCorridorHighestROIThisQuarter();
    if (!bestCorridor) return null;

    const plan: ModernizationPlan = {
      id: `PLAN-${Date.now()}`,
      corridorId: bestCorridor.id,
      budgetUSD: bestCorridor.attrs.recommendedBudgetUSD ?? 10_000_000,
      expectedROI: bestCorridor.attrs.roiThisQuarter ?? 0.25,
      steps: this.lifecycleEngine["lifecycle"].steps.map(s => s.id as LifecycleStepId)
    };

    const now = new Date().toISOString();
    await this.repo.save({
      id: plan.id,
      corridorId: plan.corridorId,
      budgetUSD: plan.budgetUSD,
      expectedROI: plan.expectedROI,
      steps: plan.steps,
      status: "PENDING",
      createdAt: now,
      updatedAt: now
    });

    return plan;
  }

  async executePlan(planId: string, context: any): Promise<void> {
    const plans = await this.repo.getActivePlans();
    const plan = plans.find(p => p.id === planId);
    if (!plan) throw new Error(`Plan not found: ${planId}`);

    await this.repo.updateStatus(plan.id, "RUNNING");

    try {
      for (const stepId of plan.steps) {
        await this.lifecycleEngine.startStep(stepId, {
          ...context,
          corridorId: plan.corridorId,
          budgetUSD: plan.budgetUSD
        });
      }
      await this.repo.updateStatus(plan.id, "COMPLETED");
    } catch (err) {
      await this.repo.updateStatus(plan.id, "FAILED");
      throw err;
    }
  }
}

// smb/core/gmig/GMIGService.ts
import { Pool } from "pg";
import { FederationTwinState } from "../../core/twin/TwinFederationTypes";

export class GMIGService {
  constructor(private db: Pool) {}

  async ingestFederationTwin(fed: FederationTwinState, ctx: { quarter: string }) {
    for (const nation of fed.nations) {
      for (const c of nation.corridors) {
        await this.db.query(
          `INSERT INTO gmig_nodes (id,type,attrs)
           VALUES ($1,'CORRIDOR',$2)
           ON CONFLICT (id) DO UPDATE SET attrs=$2, updated_at=NOW()`,
          [
            c.corridorId,
            {
              roiThisQuarter: c.roiPercent,
              nationId: nation.nationId,
              quarter: ctx.quarter,
              throughput: c.throughputVehicles,
              crossBorder: c.crossBorder ?? false
            }
          ]
        );
      }
    }
  }

  async queryTopCorridorsByROI(limit: number) {
    const { rows } = await this.db.query(
      `SELECT id, attrs
       FROM gmig_nodes
       WHERE type='CORRIDOR' AND attrs ? 'roiThisQuarter'
       ORDER BY (attrs->>'roiThisQuarter')::float DESC
       LIMIT $1`,
      [limit]
    );
    return rows;
  }

  // GMQL example: corridors with ROI > threshold and cross‑border treaties
  async gmqlCorridorsHighROIWithTreaties(minROI: number) {
    const { rows } = await this.db.query(
      `SELECT n.id, n.attrs
       FROM gmig_nodes n
       WHERE n.type='CORRIDOR'
         AND (n.attrs->>'roiThisQuarter')::float >= $1
         AND (n.attrs->>'crossBorder')::boolean = true`,
      [minROI]
    );
    return rows;
  }
}

// smb/core/amp/AMPService.ts
import { Pool } from "pg";

    const plans = [];

    for (const c of corridors) {
      const roi = parseFloat(c.attrs.roiThisQuarter);
      let budget = 10_000_000;

      // AMP‑Policies
      if (roi > 0.3) budget *= 1.1; // auto‑increase budget
      if (roi < 0.1) continue; // skip low‑ROI corridors

      const id = `PLAN-${ctx.quarter}-${c.id}`;
      await this.db.query(
        `INSERT INTO modernization_plans
         (id,corridor_id,budget_usd,expected_roi,steps,status)
         VALUES ($1,$2,$3,$4,$5,'PENDING')
         ON CONFLICT (id) DO NOTHING`,
        [id, c.id, budget, roi, ["STEP-DETECT","STEP-DESIGN","STEP-FUND","STEP-EXECUTE","STEP-MEASURE","STEP-REBALANCE"]]
      );

import { Pool } from "pg";

export type SMBEntityType = "CITY" | "STATE" | "NATION" | "FEDERATION";

export interface SMBEntity {
  id: string;
  name: string;
  type: SMBEntityType;
  parentId?: string;
  sovereignId?: string;
  federationId?: string;
  status: "PENDING" | "ACTIVE" | "FAILED";

export class SMBEntityRegistry {
  constructor(private db: Pool) {}

  async register(entity: SMBEntity): Promise<void> {
    await this.db.query(
      `INSERT INTO smb_entities
       (id,name,type,parent_id,sovereign_id,federation_id,status)
       VALUES ($1,$2,$3,$4,$5,$6,$7)
       ON CONFLICT (id) DO UPDATE SET
         name=$2,type=$3,parent_id=$4,sovereign_id=$5,
         federation_id=$6,status=$7,updated_at=NOW()`,
      [
        entity.id,
        entity.name,
        entity.type,
        entity.parentId ?? null,
        entity.sovereignId ?? null,
        entity.federationId ?? null,
        entity.status
      ]
    );
  }

  async setStatus(id: string, status: SMBEntity["status"]): Promise<void> {
    await this.db.query(
      `UPDATE smb_entities SET status=$2, updated_at=NOW() WHERE id=$1`,
      [id, status]
    );
  }

  async get(id: string): Promise<SMBEntity | null> {
    const { rows } = await this.db.query(
      `SELECT id,name,type,parent_id,sovereign_id,federation_id,status
       FROM smb_entities WHERE id=$1`,
      [id]
    );
    if (!rows[0]) return null;
    const r = rows[0];
    return {
      id: r.id,
      name: r.name,
      type: r.type,
      parentId: r.parent_id ?? undefined,
      sovereignId: r.sovereign_id ?? undefined,
      federationId: r.federation_id ?? undefined,

// smb/core/onboarding/SMBOnboardingPipeline.ts
import { SMBEntityRegistry, SMBEntityType } from "./SMBEntityRegistry";
import { TwinFederationService } from "../twin/TwinFederationService";
import { GMIGService } from "../gmig/GMIGService";
import { AMPService } from "../amp/AMPService";
import { StabilityService } from "../stability/StabilityService";

import { CorridorGuardianService } from "../corridor/CorridorGuardianService";
import { GMSXService } from "../simulation/GMSXService";
import { SMBReasoningEngine } from "../ai/SMBReasoningEngine";
import { SMBConstitutionEnforcer } from "../governance/SMBConstitutionEnforcer";
import { SMBFederationCoordinator } from "../federation/SMBFederationCoordinator";

export interface OnboardingRequest {
  id: string;
  name: string;
  type: SMBEntityType;
  parentId?: string;
  sovereignId?: string;
  federationId?: string;
  quarter: string;
}

export interface OnboardingResult {
  id: string;
  type: SMBEntityType;
  status: "ACTIVE" | "FAILED";
  details: Record<string, any>;
}

export class SMBOnboardingPipeline {
  constructor(
    private registry: SMBEntityRegistry,
    private twinFed: TwinFederationService,
    private gmig: GMIGService,
    private amp: AMPService,
    private stability: StabilityService,
    private marketplace: MarketplaceService,
    private treatyAI: TreatyAIService,
    private corridorGuardian: CorridorGuardianService,
    private gmsx: GMSXService,
    private reasoning: SMBReasoningEngine,
    private constitution: SMBConstitutionEnforcer,
    private federationCoord: SMBFederationCoordinator
  ) {}

  async onboard(req: OnboardingRequest): Promise<OnboardingResult> {
    // 1. register entity
    await this.registry.register({
      id: req.id,
      name: req.name,
      type: req.type,
      parentId: req.parentId,
      sovereignId: req.sovereignId,
      federationId: req.federationId,
      status: "PENDING"
    });

    const ctx = {
      sovereignId: req.sovereignId ?? req.id,
      federationId: req.federationId ?? `FED-${req.id}`,
      quarter: req.quarter
    };

    try {
      // 2. initialize digital twin (city/state/nation or federation)
      const federationTwin =
        req.type === "FEDERATION"
          ? await this.twinFed.buildFederation(ctx.federationId!, [ctx.sovereignId])
          : await this.twinFed.buildFederation(ctx.federationId!, [ctx.sovereignId]);

      // 3. connect to GMIG (ingest twin)
      await this.gmig.ingestFederationTwin(federationTwin, ctx);

      // 4. load constitution + policies (evaluate baseline)
      const constitutionResult = await this.constitution.evaluate(ctx);

      // 5. configure treasury + SCX pools (via StabilityService)
      await this.stability.initializePoolsForEntity(req.id, ctx);

      // 6. activate modernization lifecycle + AMP planning
      const plans = await this.amp.generateQuarterPlans(ctx);

      // 7. enable marketplace (seed instruments)
      await this.marketplace.seedInitialInstrumentsForEntity(req.id, ctx);

      // 8. enable Corridor Guardian (start watchers)
      await this.corridorGuardian.ensureGuardiansRunning(ctx);

      // 9. enable TreatyAI (if nation/federation)
      if (req.type === "NATION" || req.type === "FEDERATION") {
        await this.treatyAI.initializeForEntity(req.id, ctx);
      }

      // 10. enable GMSX simulation (baseline scenario)
      const sim = await this.gmsx.runQuarterScenario(ctx, federationTwin);

      // 11. run AI reasoning + federation coordination
      const insight = await this.reasoning.reasonQuarter(ctx);
      const fedOpt = await this.federationCoord.optimizeFederation(ctx);

      // 12. mark entity ACTIVE
      await this.registry.setStatus(req.id, "ACTIVE");

      return {
        id: req.id,
        type: req.type,
        status: "ACTIVE",
        details: {
          constitution: constitutionResult,
          plans,
          simulation: sim,
          insight,
          federationOptimization: fedOpt
        }
      };
    } catch (err) {
      await this.registry.setStatus(req.id, "FAILED");
      return {
        id: req.id,
        type: req.type,
        status: "FAILED",
        details: { error: (err as Error).message }
      };
    }
  }
}

import { SMBEntityRegistry } from "../onboarding/SMBEntityRegistry";
import { FabricOrchestrator } from "../FabricOrchestrator";

export class GlobalFabricScheduler {
  constructor(
    private registry: SMBEntityRegistry,
    private fabric: FabricOrchestrator
  ) {}

  async runGlobalTick() {
    const entities = await this.registry.listActive();
    const quarter = this.currentQuarter();

    for (const e of entities) {
      await this.fabric.runCycle({
        entityId: e.id,
        entityType: e.type,
        sovereignId: e.sovereignId ?? e.id,
        federationId: e.federationId ?? `FED-${e.id}`,
        quarter
      });
    }
  }

  currentQuarter(): string {
    const d = new Date();
    const q = Math.floor(d.getMonth() / 3) + 1;
    return `${d.getFullYear()}-Q${q}`;
  }
}
