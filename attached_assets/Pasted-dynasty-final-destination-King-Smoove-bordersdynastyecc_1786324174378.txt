3. NIMS compliance engine (technical, not legal)
ts
// nims/core/NIMSComplianceEngine.ts
import { NIMSConstitution, NIMSPrinciple } from "./NIMSConstitution";
import { NIMSTelemetryEnvelope } from "./NIMSDataStandard";

export interface ComplianceResult {
  compliant: boolean;
  score: number; // 0–1
  violations: { principleId: string; reason: string }[];
}

export class NIMSComplianceEngine {
  constructor(private constitution: NIMSConstitution) {}

  evaluateTelemetry(env: NIMSTelemetryEnvelope): ComplianceResult {
    const violations: { principleId: string; reason: string }[] = [];

    // Example: data minimization check
    const dataSize = JSON.stringify(env.payload).length;
    const maxSize = 1024 * 64; // 64KB
    if (dataSize > maxSize) {
      violations.push({
        principleId: "DATA_MINIMIZATION",
        reason: "Telemetry payload exceeds recommended size."
      });
    }

    const score =
      1 - violations.length / Math.max(this.constitution.principles.length, 1);

    return {
      compliant: violations.length === 0,
      score,
      violations
    };
  }
}

7. Algorithmic inclusion engine
ts
// inclusion/core/InclusionScore.ts
export interface InclusionScore {
  entityId: string;
  dataDensity: number; // 0–1
  modernizationAccess: number; // 0–1
  riskOfExclusion: number; // 0–1
}

export class InclusionEngine {
  computeScore(entityId: string, metrics: {
    dataPoints: number;
    population: number;
    modernizationProjects: number;
  }): InclusionScore {
    const dataDensity = Math.min(metrics.dataPoints / Math.max(metrics.population, 1), 1);
    const modernizationAccess = Math.min(metrics.modernizationProjects / 10, 1);
    const riskOfExclusion = 1 - (dataDensity * 0.5 + modernizationAccess * 0.5);

    return {
      entityId,
      dataDensity,
      modernizationAccess,
      riskOfExclusion
    };
  }

  needsFallback(score: InclusionScore): boolean {
    return score.riskOfExclusion > 0.7;
  }
}

import { FabricOrchestrator } from "./FabricOrchestrator";

// NIMS + privacy + resilience + inclusion + oversight
import { DefaultNIMSConstitution } from "../nims/core/NIMSConstitution";
import { NIMSComplianceEngine } from "../nims/core/NIMSComplianceEngine";
import { TelemetryPrivacyService } from "../privacy/core/TelemetryPrivacyService";
import { TelemetryGuard } from "../resilience/core/TelemetryGuard";
import { RollbackManager } from "../resilience/core/RollbackManager";
import { InclusionEngine } from "../inclusion/core/InclusionEngine";
import { ExplainabilityService } from "../transparency/core/ExplainabilityService";
import { OptimizationLogStore } from "../transparency/core/OptimizationLog";
import { HumanReviewQueue } from "../oversight/core/HumanReviewQueue";

export interface FabricInfra {
  fabric: FabricOrchestrator;
  nimsCompliance: NIMSComplianceEngine;
  privacy: TelemetryPrivacyService;
  telemetryGuard: TelemetryGuard;
  rollback: RollbackManager;
  inclusion: InclusionEngine;
  explainability: ExplainabilityService;
  logStore: OptimizationLogStore;
  reviewQueue: HumanReviewQueue;
}

export function buildFabricInfra(pool: Pool, fabric: FabricOrchestrator): FabricInfra {
  const nimsCompliance = new NIMSComplianceEngine(DefaultNIMSConstitution);

  const privacy = new TelemetryPrivacyService({
    differentialPrivacyEpsilon: 1.0,
    anonymizationEnabled: true,
    dataMinimizationEnabled: true
  });

  const telemetryGuard = new TelemetryGuard();
  const rollback = new RollbackManager();
  const inclusion = new InclusionEngine();
  const explainability = new ExplainabilityService();

  const logStore: OptimizationLogStore = {
    async append(decision) {
      await pool.query(
        `INSERT INTO optimization_logs
         (id, ts, sovereign_id, input_summary, output_summary, metrics, explanation)
         VALUES ($1,$2,$3,$4,$5,$6,$7)`,
        [
          decision.id,
          decision.timestamp,
          decision.sovereignId,
          decision.inputSummary,
          decision.outputSummary,
          JSON.stringify(decision.metrics),
          decision.explanation
        ]
      );
    },
    async listBySovereign(sovereignId) {
      const { rows } = await pool.query(
        `SELECT * FROM optimization_logs WHERE sovereign_id=$1 ORDER BY ts DESC`,
        [sovereignId]
      );
      return rows.map(r => ({
        id: r.id,
        timestamp: r.ts,
        sovereignId: r.sovereign_id,
        inputSummary: r.input_summary,
        outputSummary: r.output_summary,
        metrics: r.metrics,
        explanation: r.explanation
      }));
    }
  };

  const reviewQueue = new HumanReviewQueue();

  return {
    fabric,
    nimsCompliance,
    privacy,
    telemetryGuard,
    rollback,
    inclusion,
    explainability,
    logStore,
    reviewQueue
  };
}

3. Minimal end‑to‑end flow
Telemetry → Privacy → Compliance → Optimization → Logging → Oversight
ts
// src/core/FabricSafePipeline.ts
import { FabricInfra } from "./FabricInfrastructure";
import { NIMSTelemetryEnvelope } from "../nims/core/NIMSDataStandard";
import { FabricContext } from "./FabricTypes";

export class FabricSafePipeline {
  constructor(private infra: FabricInfra) {}

  async processTelemetry(env: NIMSTelemetryEnvelope, ctx: FabricContext) {
    // 1. Resilience: detect poisoning
    const anomaly = this.infra.telemetryGuard.detectPoisoning(env);
    if (anomaly) {
      // snapshot for rollback, flag for human review
      await this.infra.rollback.createSnapshot(ctx.sovereignId);
      await this.infra.reviewQueue.enqueue({
        id: anomaly.id,
        sovereignId: ctx.sovereignId,
        decisionId: "",
        reason: anomaly.reason,
        status: "OPEN"
      });
      return { status: "ANOMALY_DETECTED", anomaly };
    }

    // 2. Privacy: anonymize / minimize
    const anon = this.infra.privacy.anonymize(env);

    // 3. NIMS compliance: check envelope
    const compliance = this.infra.nimsCompliance.evaluateTelemetry(env);
    if (!compliance.compliant) {
      await this.infra.reviewQueue.enqueue({
        id: `nims-${Date.now()}`,
        sovereignId: ctx.sovereignId,
        decisionId: "",
        reason: `NIMS violations: ${compliance.violations
          .map(v => v.reason)
          .join(", ")}`,
        status: "OPEN"
      });
    }

    // 4. Optimization: run normal fabric cycle using anonymized data
    const result = await this.infra.fabric.runCycle(ctx);

    // 5. Transparency: log decision
    const explanation = this.infra.explainability.generateExplanation(
      "FabricOptimizer",
      { telemetry: anon, ctx },
      result
    );

    await this.infra.logStore.append({
      id: `opt-${Date.now()}`,
      timestamp: new Date().toISOString(),
      sovereignId: ctx.sovereignId,
      inputSummary: `Telemetry from ${env.sourceType}`,
      outputSummary: `Modernization cycle executed`,
      metrics: { complianceScore: compliance.score },
      explanation
    });

    // 6. Oversight: if compliance score low, open case
    if (compliance.score < 0.7) {
      await this.infra.reviewQueue.enqueue({
        id: `oversight-${Date.now()}`,
        sovereignId: ctx.sovereignId,
        decisionId: `opt-${Date.now()}`,
        reason: "Low NIMS compliance score",
        status: "OPEN"
      });
    }

    return {
      status: "OK",
      compliance,
      result
    };
  }
}

3. RBAC guard middleware for API
ts
// rbac/api/RbacMiddleware.ts
import { Request, Response, NextFunction } from "express";
import { FabricRole, RolePermissions } from "../core/Permissions";

interface UserContext {
  id: string;
  role: FabricRole;
  scope: { sovereignId?: string; federationId?: string };
}

declare module "express-serve-static-core" {
  interface Request {
    user?: UserContext;
  }
}

export function requirePermission(resource: string, action: "READ" | "WRITE" | "EXECUTE") {
  return (req: Request, res: Response, next: NextFunction) => {
    const user = req.user;
    if (!user) return res.status(401).json({ error: "unauthenticated" });

    const perms = RolePermissions[user.role] ?? [];
    const allowed = perms.some(p => p.resource === resource && p.action === action);
    if (!allowed) return res.status(403).json({ error: "forbidden" });

    next();
  };
}

# DynastyOS / Fabric Operator Runbooks
Technical Handbook (Non‑Political)

This handbook defines the operational responsibilities, access scopes, and
expected workflows for all roles interacting with the Fabric Safety Layer:
Transparency, Inclusion, Privacy, Resilience, and Oversight.

It is strictly technical and focused on modernization infrastructure.

---

## 1. Role Overview

DynastyOS uses four operator roles:

- **FABRIC_ADMIN** — full cluster access, all sovereigns/federations
- **FABRIC_OPERATOR** — operational control within assigned scope
- **FABRIC_AUDITOR** — read‑only verification and compliance checks
- **FABRIC_VIEWER** — limited read‑only visibility

Each role interacts with the Fabric Safety Console according to its permissions.

---

## 2. FABRIC_ADMIN Runbook

### Scope
Full access across all sovereigns and federations.

### Responsibilities
- Maintain Fabric Safety Layer configuration:
  - Privacy parameters (DP epsilon, anonymization rules)
  - Resilience thresholds (poisoning heuristics, rollback triggers)
  - Inclusion scoring weights
- Manage optimization logs and transparency outputs
- Trigger rollback snapshots during anomalies
- Approve updates to NIMS technical parameters
- Coordinate incident response with operators

### Daily Tasks
- Review cluster‑wide optimization logs
- Check resilience alerts and telemetry anomalies
- Validate inclusion scores for underserved regions
- Ensure privacy layer is functioning (anonymization + minimization)
- Monitor oversight queue for escalated cases

### Escalation
Admins escalate only when:
- Multiple sovereigns show telemetry poisoning
- Rollback fails or produces inconsistent state hashes
- Inclusion risk > 0.8 across multiple entities

---

## 3. FABRIC_OPERATOR Runbook

### Scope
Assigned sovereign(s) or federation(s).

### Responsibilities
- Operate the Fabric Safety Console for their scope
- Run privacy anonymization on telemetry samples
- Execute resilience checks on suspicious telemetry
- Open oversight cases when compliance or inclusion scores drop
- Monitor optimization logs for anomalies

### Daily Tasks
- Check optimization logs for unexpected patterns
- Compute inclusion scores for low‑data or underserved regions
- Run telemetry poisoning checks on corridor/logistics streams
- Trigger fallback planning when inclusion risk > 0.7
- Submit oversight cases for human review

### Escalation
Operators escalate when:
- Compliance score < 0.7
- Inclusion risk > 0.7
- Telemetry anomalies repeat within 24 hours

---

## 4. FABRIC_AUDITOR Runbook

### Scope
Read‑only across assigned sovereign/federation scope.

### Responsibilities
- Verify transparency outputs:
  - Optimization logs exist
  - Explanations are human‑readable
- Validate inclusion scoring:
  - Data density and access metrics are consistent
- Confirm privacy and resilience workflows are invoked
- Produce technical audit reports on system behavior

### Daily Tasks
- Review logs for missing explanations
- Check inclusion scores for consistency
- Confirm anonymization is applied to telemetry samples
- Validate resilience alerts and rollback snapshots

### Escalation
Auditors escalate when:
- Logs lack explanations
- Inclusion scores appear stale or inconsistent
- Privacy layer fails to anonymize telemetry

---

## 5. FABRIC_VIEWER Runbook

### Scope
Limited read‑only visibility.

### Responsibilities
- Observe high‑level transparency and inclusion dashboards
- Provide feedback on clarity and usability
- No operational actions

### Daily Tasks
- Monitor dashboards for situational awareness
- Report UI or clarity issues to operators

### Escalation
Viewers do not escalate operational issues.

### Transparency Module
- Optimization logs

### Inclusion Module
- Inclusion scoring engine

### End‑to‑End Flow
1. **Telemetry arrives**
2. **Resilience check**

3. **Privacy layer**

# Fabric Safety Console
Technical Overview (Transparency, Inclusion, Privacy, Resilience)

The Fabric Safety Console is the primary operator interface for monitoring and
controlling the safety layer of DynastyOS/Fabric. It exposes four core panels:

- Transparency Panel
- Inclusion Panel
- Privacy Panel
- Resilience Panel

All functionality is technical and focused on modernization infrastructure.

---

## 1. Transparency Panel

### Purpose
Provide human‑readable visibility into Fabric optimization decisions.

### Features
- List of optimization logs per sovereign
- Timestamped decision summaries
- Human‑readable explanations from the Explainability Service
- Basic metrics (compliance score, inclusion risk)

### Operator Actions
- Filter logs by sovereign or time range
- Inspect explanations for unexpected behavior
- Flag decisions for oversight review (via Oversight Module)

---

## 2. Inclusion Panel

### Purpose
Monitor equity and access to modernization across entities.

### Features
- Inclusion score per entity (city, corridor, region)
- Data density and modernization access metrics
- Risk of exclusion indicator

### Operator Actions
- Compute inclusion scores for target entities
- Identify high‑risk exclusion zones
- Trigger fallback planning for low‑data regions

---

## 3. Privacy Panel

### Purpose
Verify telemetry anonymization and data minimization behavior.

### Features
- Sample telemetry anonymization requests
- View anonymized payloads
- Confirm removal of identifiers and reduction to aggregate metrics

### Operator Actions
- Submit sample telemetry envelopes for anonymization
- Validate anonymization output against privacy expectations
- Report anomalies to Fabric Admins

---

## 4. Resilience Panel

### Purpose
Detect telemetry anomalies and potential poisoning.

### Features
- Telemetry poisoning checks
- Anomaly reports (reason, severity)
- Status indicators for resilience workflows

### Operator Actions
- Run checks on suspicious telemetry streams
- Initiate incident workflows (rollback + oversight)
- Monitor resilience alerts for repeated anomalies

---

## 5. Access Control

The Fabric Safety Console is governed by RBAC:

- **FABRIC_ADMIN** — full access to all panels and actions
- **FABRIC_OPERATOR** — full access within assigned scope
- **FABRIC_AUDITOR** — read‑only access to Transparency, Inclusion, Oversight
- **FABRIC_VIEWER** — read‑only access to high‑level Transparency and Inclusion

---

## 6. Integration Points

The console communicates with:

- `transparency-service` for optimization logs
- `inclusion-service` for inclusion scores
- `privacy-service` for telemetry anonymization
- `resilience-service` for telemetry checks
- `oversight-service` for review queue operations

All calls are routed through the Fabric API Gateway.

# NIMS Technical Specification
National Intelligent Modernization Standard (Technical Layer Only)

This document defines the technical aspects of NIMS as implemented in
DynastyOS/Fabric. It does not define legal, political, or policy content.

---

## 1. Purpose

NIMS provides a standardized technical framework for:

- Telemetry envelopes
- Modernization events
- Compliance checks
- Transparency protocols
- Inclusion and safety integration

---

## 2. Telemetry Envelope Specification

### Structure
- `sourceId` — unique identifier of telemetry source
- `sourceType` — `CORRIDOR | LOGISTICS | FUEL | TREASURY | SCX | TWIN`
- `sovereignId` — identifier of sovereign entity
- `federationId` — optional federation identifier
- `timestamp` — ISO 8601 timestamp
- `payload` — structured telemetry data
- `schemaVersion` — version of telemetry schema

---

## 3. Modernization Event Specification

### Structure
- `id` — unique event identifier
- `type` — `UPGRADE | MAINTENANCE | ALERT | SIMULATION_RESULT`
- `entityId` — target entity (corridor, city, asset)
- `sovereignId` — associated sovereign
- `timestamp` — ISO 8601 timestamp
- `metadata` — additional structured data

---

## 4. NIMS Constitution (Technical Principles)

Examples:

- **DATA_MINIMIZATION** — limit telemetry payload size and fields.
- **INCLUSION_BASELINE** — ensure low‑data entities still receive planning.
- **PRIVACY_PRESERVATION** — enforce anonymization and minimization.
- **RESILIENCE_CHECKS** — require poisoning detection on critical streams.

Each principle has:

- `id`
- `name`
- `description`
- `category` (`DATA | STABILITY | INCLUSION | PRIVACY | RESILIENCE`)
- `weight` (0–1 influence in compliance scoring)

---

## 5. Compliance Engine Behavior

The NIMS Compliance Engine:

- Evaluates telemetry envelopes against technical principles.
- Produces:
  - `compliant` flag
  - `score` (0–1)
  - list of violations (principle + reason)
- Integrates with:
  - Resilience Module (for anomalies)
  - Oversight Module (for low scores)
  - Transparency Module (for logging compliance metrics)

---

## 6. Transparency Protocol

NIMS requires:

- Optimization decisions to be logged with:
  - input summary
  - output summary
  - metrics (including compliance score)
  - human‑readable explanation
- Logs to be queryable per sovereign.

---

## 7. Maturity Model (Technical)

NIMS defines technical maturity levels based on:

- Telemetry coverage
- Inclusion engine usage
- Privacy configuration completeness
- Resilience workflows enabled
- Transparency logging completeness

Levels (example):

- **Level 1** — basic telemetry + logs
- **Level 2** — privacy + resilience enabled
- **Level 3** — inclusion + oversight integrated
- **Level 4** — full safety layer with continuous monitoring

---

## 8. Federation Protocol (Technical Coordination)

NIMS Federation Protocol defines:

- Shared metrics for coordination (load, capacity, stability)
- Data formats for federation‑level reports
- Interfaces for generating technical coordination plans
  (e.g., redistribution of load or capacity across sovereigns)

No political or legal content is defined here—only technical coordination.

import { FabricSafetyConsole } from "./screens/FabricSafetyConsole";
import { DashboardHome } from "./screens/DashboardHome";
import { TelemetryView } from "./screens/TelemetryView";
import { ModernizationCyclesView } from "./screens/ModernizationCyclesView";

export function App() {
  return (
    <BrowserRouter>
      <div className="operator-portal">
        <aside className="sidebar">
          <h2>DynastyOS Operator Portal</h2>
          <nav>
            <a href="/">Overview</a>
            <a href="/telemetry">Telemetry</a>
            <a href="/cycles">Modernization Cycles</a>
            <a href="/safety">Fabric Safety Console</a>
          </nav>
        </aside>
        <main className="content">
          <Routes>
            <Route path="/" element={<DashboardHome />} />
            <Route path="/telemetry" element={<TelemetryView />} />
            <Route path="/cycles" element={<ModernizationCyclesView />} />
            <Route path="/safety" element={<FabricSafetyConsole />} />
          </Routes>
        </main>
      </div>
    </BrowserRouter>
  );
}
