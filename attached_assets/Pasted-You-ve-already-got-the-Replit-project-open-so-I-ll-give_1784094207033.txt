```ts
// services/dispatchEngine.ts
import { logCodexEvent } from './codexChain';

export async function executeDispatch(loadId: string, driverId: string) {
  const event = {
    type: 'DISPATCH_ASSIGNED',
    payload: { loadId, driverId, at: new Date().toISOString() },
  };
  await logCodexEvent(event.type, event.payload);
  return { ...event.payload, status: 'ASSIGNED' };
}
```

```ts
// api/billing/checkout.ts
import Stripe from 'stripe';
import { TIERS } from '../../billing/tiers';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY);

export default async function handler(req, res) {
  const { tier, carrierId } = req.body;

  const session = await stripe.checkout.sessions.create({
    mode: 'subscription',
    line_items: [{
      price_data: {
        currency: 'usd',
        product: TIERS[tier].id,
        recurring: { interval: 'month' },
        unit_amount: TIERS[tier].price,
      },
      quantity: 1,
    }],
    success_url: `${process.env.APP_URL}/dashboard?carrier=${carrierId}&success=true`,
    cancel_url: `${process.env.APP_URL}/pricing?cancel=true`,
  });

  res.json({ url: session.url });
}
```

### 7. Carrier signup + DoorDash‑style job feed

```tsx
// pages/signup/carrier.tsx
export default function CarrierSignup() {
  async function handleSubmit(e) {
    e.preventDefault();
    const form = new FormData(e.currentTarget);
    const body = Object.fromEntries(form.entries());

    const res = await fetch('/api/carrier/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    });
    const { carrierId } = await res.json();

    // redirect to pricing / Stripe
    window.location.href = `/pricing?carrier=${carrierId}`;
  }

  return (
    <main className="min-h-screen bg-slate-950 text-white p-6">
      <h1 className="text-xl mb-4">Owner‑Operator Onboarding</h1>
      <p className="text-xs text-gray-300 mb-4">
        Box trucks & Class A drivers must sign up here to receive loads.
      </p>
      <form onSubmit={handleSubmit} className="space-y-3">
        <input name="companyName" placeholder="Company Name" required />
        <input name="email" placeholder="Email" required />
        <select name="equipment" required>
          <option value="BOX_TRUCK">Box Truck</option>
          <option value="CLASS_A">Class A Tractor</option>
          <option value="SPRINTER">Sprinter / Cargo Van</option>
        </select>
        <button type="submit" className="bg-purple-600 px-4 py-2 rounded">
          Continue
        </button>
      </form>
    </main>
  );
}
```
