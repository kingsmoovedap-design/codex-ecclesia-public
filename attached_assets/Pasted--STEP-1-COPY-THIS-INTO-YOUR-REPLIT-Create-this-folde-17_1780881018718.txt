# Initialize memory + tools
memory = Memory()

tools = {
    "logistics": LogisticsTools(memory),
    "reverse": ReverseTools(memory),
    "warehouse": WarehouseTools(memory),
    "dispatch": DispatchTools(memory),
    "governance": GovernanceTools(memory),
    "compliance": ComplianceTools(memory),

def think(task):
    """
    Core reasoning engine for DivinityVX.
    Decides which tool to use and how to respond.
    """

    memory.store_interaction("task_received", task)

    category = task.get("category")
    action = task.get("action")
    payload = task.get("payload", {})

    if category not in tools:
        return {
            "status": "error",
            "message": f"Unknown category '{category}'",
            "received": task
        }

    tool = tools[category]

    if not hasattr(tool, action):
        return {
            "status": "error",
            "message": f"Unknown action '{action}' for category '{category}'",
            "received": task
        }

    try:
        result = getattr(tool, action)(payload)
        memory.store_interaction("task_completed", {"task": task, "result": result})
        return {
            "status": "success",
            "category": category,
            "action": action,
            "result": result
        }
    except Exception as e:
        memory.store_interaction("task_error", {"task": task, "error": str(e)})
        return {
            "status": "error",
            "message": str(e),
            "task": task
        }

def main():
    """Reads JSON input from Node.js and returns JSON output."""
    raw = sys.stdin.read()
    try:
        task = json.loads(raw)
    except:
        print(json.dumps({"error": "Invalid JSON input"}))
        return

    response = think(task)
    print(json.dumps(response))

if __name__ == "__main__":
    Main()

"""
DivinityVX Memory Engine
Stores short-term and long-term memory for the AI system.
This is NOT consciousness — it is structured state storage.
"""

import json
import os
from datetime import datetime

MEMORY_FILE = "divinityvx_memory.json"

class Memory:
    def __init__(self):
        self.short_term = []
        self.long_term = []
        self.load()

    def load(self):
        """Load memory from disk."""
        if os.path.exists(MEMORY_FILE):
            try:
                with open(MEMORY_FILE, "r") as f:
                    data = json.load(f)
                    self.short_term = data.get("short_term", [])
                    self.long_term = data.get("long_term", [])
            except:
                self.short_term = []
                self.long_term = []

    def save(self):
        """Persist memory to disk."""
        data = {
            "short_term": self.short_term[-200:],  # keep last 200 interactions
            "long_term": self.long_term
        }
        with open(MEMORY_FILE, "w") as f:
            json.dump(data, f, indent=2)

    def store_interaction(self, event_type, payload):
        """Store an interaction in short-term memory."""
        entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "type": event_type,
            "payload": payload
        }
        self.short_term.append(entry)
        if len(self.short_term) > 200:
            self.short_term.pop(0)
        self.save()

    def store_long_term(self, category, data):
        """Store long-term insights or patterns."""
        entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "category": category,
            "data": data
        }
        self.long_term.append(entry)
        self.save()

    def recall_recent(self, limit=10):
        """Return the most recent memory entries."""
        return self.short_term[-limit:]

    def recall_category(self, category):
        """Return long-term memory entries for a category."""
        return [m for m in self.long_term if m["category"] == category]

class DVEventBus {
  constructor() {
    this.listeners = {};
    this.poll();
  }

  on(event, callback) {
    if (!this.listeners[event]) this.listeners[event] = [];
    this.listeners[event].push(callback);
  }

  emit(event, data) {
    if (this.listeners[event]) {
      this.listeners[event].forEach(cb => cb(data));
    }
  }

  async poll() {
    const res = await fetch("/api/codex/events");
    const data = await res.json();

    data.events.forEach(evt => {
      this.emit(evt.type, evt);
    });

    setTimeout(() => this.poll(), 3000);
  }
}

window.DVBus = new DVEventBus();

⭐ 2. DRIVER ETA PREDICTION
This uses Haversine distance + average speed to estimate arrival time.
📌 Backend — Add to /backend/logistics/driver.js
function haversine(lat1, lon1, lat2, lon2) {
  const R = 6371;
  const dLat = (lat2-lat1) * Math.PI/180;
  const dLon = (lon2-lon1) * Math.PI/180;
  const a = Math.sin(dLat/2)**2 +
            Math.cos(lat1*Math.PI/180) *
            Math.cos(lat2*Math.PI/180) *
            Math.sin(dLon/2)**2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

router.post('/eta', (req, res) => {
  const { driverId, destLat, destLng } = req.body;
  const pos = gpsPositions[driverId];

  if (!pos) return res.json({ error: "Driver not found" });

  const distance = haversine(pos.lat, pos.lng, destLat, destLng);
  const avgSpeed = 55; // mph or 88 km/h
  const etaHours = distance / avgSpeed;

  res.json({
    driverId,
    distanceKm: distance.toFixed(2),
    etaHours: etaHours.toFixed(2),
    etaReadable: `${Math.floor(etaHours)}h ${Math.round((etaHours%1)*60)}m`
  });
});

class MarketplaceTools:
    def __init__(self, memory):
        self.memory = memory

    def recommendBid(self, payload):
        load = payload.get("load", {})
        bids = payload.get("bids", [])

        base = load.get("declaredValue", 0) * 0.12
        competition = len([b for b in bids if b["loadId"] == load.get("id")])
        adjustment = -(competition * 0.5)

        recommended = max(50, base + adjustment)

        result = {
            "recommendedBid": round(recommended, 2),
            "competition": competition,
            "reasoning": "Based on load value, competition, and historical patterns."
        }

        self.memory.store_interaction("marketplace_recommend", result)
        return result

      if (dist > 0.3) anomalies.push({ type: "gps_jump", driverId: p.driverId });
    }
    p.last = { lat: p.lat, lng: p.lng };
  });

  // Bid spikes
  const grouped = {};
  bids.forEach(b => {
    if (!grouped[b.loadId]) grouped[b.loadId] = [];
    grouped[b.loadId].push(b.amount);
  });

  Object.keys(grouped).forEach(loadId => {
    const arr = grouped[loadId];
    const avg = arr.reduce((a,b)=>a+b)/arr.length;

  });
