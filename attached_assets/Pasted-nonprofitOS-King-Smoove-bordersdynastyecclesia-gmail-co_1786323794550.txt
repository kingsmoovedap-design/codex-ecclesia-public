-- Organizations
CREATE TABLE organizations (
  id UUID PRIMARY KEY,
  name TEXT NOT NULL,
  org_type TEXT NOT NULL, -- 'nonprofit', 'community_group', 'city', 'law_enforcement'
  location TEXT,
  capacity_level TEXT
);

-- Projects
CREATE TABLE projects (
  id UUID PRIMARY KEY,
  organization_id UUID REFERENCES organizations(id),
  title TEXT NOT NULL,
  description TEXT,
  project_type TEXT, -- 'crime_prevention', 'youth_safety', 'tech', 'community_engagement', 'neighborhood_watch'
  target_population TEXT,
  local_concerns TEXT,
  desired_outcomes TEXT,
  status TEXT DEFAULT 'draft'
);

-- Priorities (federal/state/local)
CREATE TABLE priorities (
  id UUID PRIMARY KEY,
  level TEXT NOT NULL, -- 'federal', 'state', 'local'
  agency TEXT,         -- 'DOJ', 'COPS', 'DHS', etc.
  name TEXT NOT NULL,
  description TEXT,
  tags TEXT[]          -- e.g. ['violence_reduction','youth','community_policing']
);

-- Grants
CREATE TABLE grants (
  id UUID PRIMARY KEY,
  name TEXT NOT NULL,
  agency TEXT,
  level TEXT,          -- 'federal','state','local'
  description TEXT,
  eligibility JSONB,   -- rules for org_type, project_type, etc.
  priority_ids UUID[]  -- linked priorities
);

-- Partnerships
CREATE TABLE partnerships (
  id UUID PRIMARY KEY,
  project_id UUID REFERENCES projects(id),
  partner_org_id UUID REFERENCES organizations(id),
  role TEXT,           -- 'lead','subrecipient','law_enforcement_partner','city_partner'
  mou_url TEXT,
  status TEXT          -- 'draft','active','completed'
);

-- Case studies
CREATE TABLE case_studies (
  id UUID PRIMARY KEY,
  title TEXT,
  category TEXT,       -- 'violence_reduction','youth_safety','community_policing',...
  summary TEXT,
  success_factors TEXT,
  funding_sources TEXT,
  replicable_components TEXT
);

-- Compliance & reporting
CREATE TABLE reports (
  id UUID PRIMARY KEY,
  project_id UUID REFERENCES projects(id),
  period_start DATE,
  period_end DATE,
  data JSONB,
  submitted_at TIMESTAMP
);

1. Eligibility Pathway Service

// services/eligibilityService.ts
export function recommendPathway(project: Project, org: Organization) {
  const { project_type } = project;
  const { org_type } = org;

  if (
    ['crime_prevention', 'youth_safety', 'community_engagement', 'tech', 'neighborhood_watch']
      .includes(project_type)
  ) {
    return 'Option A'; // Partner with law enforcement
  }

  if (org_type === 'nonprofit') {
    return 'Option B'; // Apply as nonprofit
  }

  if (org_type === 'community_group') {
    return 'Option C'; // Apply through city/county
  }

  return 'Option B'; // default fallback
}
2. Priority Alignment Service

// services/priorityAlignmentService.ts
export async function alignProjectToPriorities(projectId: string) {
  const project = await db.projects.findById(projectId);
  const priorities = await db.priorities.findByTags(project.project_type, project.target_population);

  const grants = await db.grants.findByPriorities(priorities.map(p => p.id));

  return {
    priorities,
    grants
  };
}

// services/grantPackageService.ts
export async function generateGrantPackage(projectId: string) {
  const project = await db.projects.findById(projectId);
  const partnerships = await db.partnerships.findByProject(projectId);

  return {
    needsAssessment: `Based on local concerns: ${project.local_concerns}`,
    problemStatement: `This project addresses ${project.project_type} issues in ${project.location}`,
    narrative: `The project titled "${project.title}" aims to ${project.desired_outcomes}...`,
    budgetTemplate: { lineItems: [], total: 0 },
    sustainabilityPlan: `Sustainability will be achieved through ongoing partnerships and diversified funding.`,
    partnerships
  };
}

import * as eligibilityService from '../services/eligibility.service';

  const result = await caseStudiesService.listCaseStudies();

  const { projectId } = req.params;

Example service (src/server/services/eligibility.service.ts)
ts
import { Project } from '../models/Project.model';
import { Organization } from '../models/Organization.model';
import db from '../db';

export async function getEligibilityForProject(projectId: string) {
  const project = await db.projects.findById(projectId) as Project;
  const org = await db.organizations.findById(project.organization_id) as Organization;

  let pathway: 'Option A' | 'Option B' | 'Option C' = 'Option B';

  if (
    ['crime_prevention', 'youth_safety', 'community_engagement', 'tech', 'neighborhood_watch']
      .includes(project.project_type)
  ) {
    pathway = 'Option A';
  } else if (org.org_type === 'community_group') {
    pathway = 'Option C';
  }

  return {
    pathway,
    requiredDocuments: [],
    requiredPartners: [],
    grantCategoriesUnlocked: []
  };
}

export class InMemoryLifeStorage implements LifeStorage {
  private snapshots: LifeSnapshot[] = [];
  private goals: StoredGoal[] = [];
  private emotions: StoredEmotion[] = [];

  async saveSnapshot(snapshot: LifeSnapshot): Promise<void> {
    this.snapshots.push(snapshot);
  }

  async loadLatestSnapshot(userId: string): Promise<LifeSnapshot | null> {
    const list = this.snapshots.filter(s => s.userId === userId);
    return list.sort((a, b) => b.timestamp.getTime() - a.timestamp.getTime())[0] || null;
  }

  async saveGoal(goal: StoredGoal): Promise<void> {
    this.goals.push(goal);
  }

  async listGoals(userId: string): Promise<StoredGoal[]> {
    return this.goals.filter(g => g.userId === userId);
  }

  async saveEmotion(emotion: StoredEmotion): Promise<void> {
    this.emotions.push(emotion);
  }

  async listEmotions(userId: string): Promise<StoredEmotion[]> {
    return this.emotions.filter(e => e.userId === userId);
  }
}

import {
  CivilizationOSv2,
  CivilizationDomain,
  CivilizationState
} from "./CivilizationOSv2";

export class CivilizationOSv2Impl implements CivilizationOSv2 {
  private state: CivilizationState = {
    domains: [],
    stabilityIndex: 0.5,
    harmonyIndex: 0.5,
    riskIndex: 0.5
  };

  getState(): CivilizationState {
    return this.state;
  }

  registerDomain(domain: CivilizationDomain): void {
    this.state.domains.push(domain);
  }

  setDomainPriority(id: string, priority: number): void {
    const domain = this.state.domains.find(d => d.id === id);
    if (domain) domain.priority = priority;
  }

  computeHarmony(): number {
    const activeDomains = this.state.domains.filter(d => d.enabled);
    const avgPriority =
      activeDomains.reduce((sum, d) => sum + d.priority, 0) /
      (activeDomains.length || 1);

    this.state.harmonyIndex = avgPriority / 10;
    return this.state.harmonyIndex;
  }

  computeRisk(): number {
    const disabled = this.state.domains.filter(d => !d.enabled).length;
    this.state.riskIndex = disabled / (this.state.domains.length || 1);
    return this.state.riskIndex;
  }
}

    generations: [],
    successionPlans: [],
    continuityIndex: 0.5
  };

  getState(): DynastyState {
    return this.state;
  }

  addGeneration(profile: GenerationProfile): void {
    this.state.generations.push(profile);
  }

  defineSuccession(plan: SuccessionPlan): void {
    this.state.successionPlans.push(plan);
  }

  computeContinuity(): number {
    const readinessAvg =
      this.state.generations.reduce((sum, g) => sum + g.readinessScore, 0) /
      (this.state.generations.length || 1);

    this.state.continuityIndex = readinessAvg;
    return readinessAvg;
  }

  getSuccessorForTrustee(trusteeId: string): GenerationProfile | null {
    const plan = this.state.successionPlans.find(
      p => p.currentTrusteeId === trusteeId
    );
    if (!plan) return null;

    return this.state.generations.find(
      g => g.id === plan.successorTrusteeId
    ) || null;
  }
}
