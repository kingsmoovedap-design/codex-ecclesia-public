export interface StressScenario {
  throughputDropPercent?: number;
  incidentSpikePercent?: number;
  fuelConsumptionSpikePercent?: number;
  fuelCostSpikePercent?: number;
  logisticsCostIncreasePercent?: number;
}

export interface StressResult {
  modernizationCreditDemandChangePercent: number;
  corridorCreditDemandChangePercent: number;
  scxLiquidityChangePercent: number;
  reserveRatioStressPercent: number;
}

export class SovereignCreditStressTest {
  static run(scenario: StressScenario): StressResult {
    const modernizationCreditDemandChangePercent =
      -(scenario.throughputDropPercent ?? 0) * 0.5;

    const corridorCreditDemandChangePercent =
      -(scenario.incidentSpikePercent ?? 0) * 0.7;

    const scxLiquidityChangePercent =
      -(scenario.fuelCostSpikePercent ?? 0) * 0.5;

    const reserveRatioStressPercent =
      (scenario.fuelConsumptionSpikePercent ?? 0) * 0.3 +
      (scenario.logisticsCostIncreasePercent ?? 0) * 0.2;

    return {
      modernizationCreditDemandChangePercent,
      corridorCreditDemandChangePercent,
      scxLiquidityChangePercent,
      reserveRatioStressPercent
    };
  }
}

export interface ReserveSnapshot {

  deviationPercent: number;

}

export class SCXReserveStabilityEngine {
  static assess(snapshot: ReserveSnapshot): StabilityAssessment {
    const currentRatio = snapshot.balance / snapshot.backingValueUSD;
    const deviationPercent =
      ((currentRatio - snapshot.targetRatio) / snapshot.targetRatio) * 100;

    let status: StabilityAssessment["status"] = "STABLE";
    if (Math.abs(deviationPercent) > 10 && Math.abs(deviationPercent) <= 25) {
      status = "WARNING";
    } else if (Math.abs(deviationPercent) > 25) {
      status = "CRITICAL";
    }

// core/global/corridor/NationCorridorModernizationSimulator.ts
import { NationTwinState } from "../digitaltwin/NationTwin";
import { DigitalTwinEngine, TwinScenario, TwinScenarioResult } from "../digitaltwin/DigitalTwinEngine";

export interface NationModernizationScenario {
  nationId: string;
  scenarios: TwinScenario[];
}

export interface NationModernizationResult {
  nationId: string;
  timestamp: string;
  corridorResults: TwinScenarioResult[];
  avgROI: number;
}

export class NationCorridorModernizationSimulator {
  static run(
    nationTwin: NationTwinState,
    nationScenario: NationModernizationScenario
  ): NationModernizationResult {
    const corridorResults: TwinScenarioResult[] = [];

    for (const scenario of nationScenario.scenarios) {
      const baseline = nationTwin.corridors.find(
        c => c.corridorId === scenario.corridorId
      );
      if (!baseline) continue;
      const result = DigitalTwinEngine.runScenario(baseline, scenario);
      corridorResults.push(result);
    }

    const avgROI =
      corridorResults.reduce((sum, r) => sum + r.roiPercent, 0) /
      (corridorResults.length || 1);

    return {
      nationId: nationTwin.nationId,
      timestamp: new Date().toISOString(),
      corridorResults,
      avgROI
    };
  }
}
