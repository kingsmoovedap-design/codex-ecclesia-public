// src/digitaltwin/GlobalLogisticsDigitalTwin.ts
import { TwinSnapshot, TwinScenarioInput, TwinScenarioResult } from "./contracts";
import { pg } from "../db/postgres";

export class GlobalLogisticsDigitalTwin {
  async captureCurrentState(): Promise<TwinSnapshot> {
    const loads = await pg.query("SELECT * FROM loads");
    const carriers = await pg.query("SELECT * FROM carriers");
    const containers = await pg.query("SELECT * FROM containers");
    const payments = await pg.query("SELECT * FROM payments");
    const credits = await pg.query("SELECT * FROM sovereign_credits");

    const state = {
      loads: loads.rows,
      carriers: carriers.rows,
      containers: containers.rows,
      payments: payments.rows,
      credits: credits.rows
    };

    const snapshot: TwinSnapshot = {
      id: `TWIN-${Date.now()}`,
      snapshotTime: new Date().toISOString(),
      state
    };

    await pg.query(
      "INSERT INTO digital_twin_states (id, snapshot_time, state) VALUES ($1,$2,$3)",
      [snapshot.id, snapshot.snapshotTime, snapshot.state]
    );

    return snapshot;
  }

  async simulate(input: TwinScenarioInput): Promise<TwinScenarioResult> {
    const latest = await pg.query(
      "SELECT * FROM digital_twin_states ORDER BY snapshot_time DESC LIMIT 1"
    );
    const baseState = latest.rows[0]?.state || {};

    // simple placeholder logic; you’ll evolve this
    return {
      impactOnLoads: baseState.loads,
      impactOnCarriers: baseState.carriers,
      impactOnFees: {},
      impactOnCredits: {},
      impactOnCongestion: {}
    };
  }
}

2. Sovereign Logistics Cloud rollout — API + auth skeleton
API key + scopes
ts
// src/cloud/SovereignLogisticsCloud.ts
import { pg } from "../db/postgres";

export class SovereignLogisticsCloud {
  async validateApiKey(apiKey: string, requiredScope: string): Promise<boolean> {
    const res = await pg.query(
      "SELECT scopes FROM api_clients WHERE api_key = $1",
      [apiKey]
    );
    if (!res.rowCount) return false;
    const scopes: string[] = res.rows[0].scopes || [];
    return scopes.includes(requiredScope);
  }
}
Express middleware
ts
// src/middleware/ApiAuth.ts
import { SovereignLogisticsCloud } from "../cloud/SovereignLogisticsCloud";

const cloud = new SovereignLogisticsCloud();

export async function apiAuth(scope: string, req, res, next) {
  const apiKey = req.headers["authorization"]?.replace("Bearer ", "");
  if (!apiKey) return res.status(401).json({ error: "missing_api_key" });

  const ok = await cloud.validateApiKey(apiKey, scope);
  if (!ok) return res.status(403).json({ error: "forbidden" });

  next();
}

// src/monetization/MonetizationOSv3.ts (extend)
async settle(ctx: FeeContext): Promise<{
  settlement: FeeSettlement;
  treasuryRoute: TreasuryRoute;
  sovereignCredit?: SovereignCredit;
}> {
  const quote = this.fees.quote(ctx);

  const settlement: FeeSettlement = {
    settlementId: `SET-${Date.now()}`,
    amount: quote.amount,
    currency: quote.currency,
    settled: true,
    timestamp: new Date().toISOString()
  };

  const treasuryRoute: TreasuryRoute = {
    id: `TR-${Date.now()}`,
    domain: ctx.domain,
    amount: quote.amount,
    currency: quote.currency,
    ledger: this.mapDomainToLedger(ctx.domain)
  };

  let sovereignCredit: SovereignCredit | undefined;

  if (ctx.domain === "REVERSE_LOGISTICS") {
    sovereignCredit = {
      id: `SC-${Date.now()}`,
      domain: ctx.domain,
      amount: quote.amount * 0.3,
      reason: "reverse_logistics_recovery",
      program: "DOT_MODERNIZATION",
      beneficiaryType: "CARRIER",
      beneficiaryId: ctx.payload?.carrierId,
      impactMetrics: { recoveredLoads: ctx.payload?.recoveredCount || 1 }
    } as any;
  }

  if (ctx.domain === "CONTAINER") {
    sovereignCredit = {
      id: `SC-${Date.now()}`,
      domain: ctx.domain,
      amount: quote.amount * 0.25,
      reason: "container_repositioning_efficiency",
      program: "PORT_DIGITALIZATION",
      beneficiaryType: "PORT",
      beneficiaryId: ctx.payload?.portId,
      impactMetrics: { reducedIdleContainers: ctx.payload?.deltaIdle || 0 }
    } as any;
  }

  return { settlement, treasuryRoute, sovereignCredit };
}
